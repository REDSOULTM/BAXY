"""Live check of the ronda-44 history trim: the real prefix in tokens, and a long session that still answers."""
import json
import os
import sys
import time
from pathlib import Path

REPO = Path(r"%USERPROFILE%\BAXY-Renew")
S = Path(r"<carpeta temporal>")
sys.path.insert(0, str(REPO / "src"))
from baxy_mind.llm_server import LlmServer  # noqa: E402
from baxy_mind import mind as mind_module  # noqa: E402
from baxy_mind.mind import Mind  # noqa: E402

server = LlmServer(r"<evaluacion>\models\Qwen3.5-4B-Q4_K_M.gguf", r"<evaluacion>\llama\llama-server.exe",
                   context=49152, log_name="trim-live.log")
server.start()
try:
    mind = Mind(server, S / "trim_live.trace.jsonl")
    catalog = Path(os.environ["LOCALAPPDATA"]) / "BAXY" / "renew" / "catalog.configure.json"
    mind.configure(json.loads(catalog.read_text(encoding="utf-8")))
    server.wait_ready(600)
    mind.warm()
    print("prefix_tokens", mind.prefix_tokens, "room", server.context - mind.prefix_tokens - mind_module.TURN_ROOM_TOKENS)
    # Seven earlier turns of ~3,300 characters each (23,100 < 24,000): the old bound kept them all.
    mind.new_session()
    filler = ("El informe trimestral dice que las ventas subieron en el norte y bajaron en el sur; " * 40)[:3200]
    for i in range(7):
        mind.conversation.append({"role": "user", "content": f"{mind_module.CLOCK_MARK} turno {i}]"})
        mind.conversation.append({"role": "user", "content": f"resume la parte {i}"})
        mind.conversation.append({"role": "assistant", "content": filler})
    whole = "".join(json.dumps(m, ensure_ascii=False) for m in mind.conversation)
    print("history chars", len(whole), "tokens", len(server.tokenize(whole)))
    t0 = time.monotonic()
    result = mind.decide("r1", "Â¿quÃ© hora es?")
    kept = sum(1 for m in mind.conversation if str(m.get("content") or "").startswith(mind_module.CLOCK_MARK))
    print("decide", round(time.monotonic() - t0, 2), "s kind", result.get("kind"), "turns kept", kept)
    print(json.dumps(result, ensure_ascii=False)[:300])
    print("first message", json.dumps(mind.conversation[0], ensure_ascii=False))
finally:
    server.close()
